#include <DHT.h>

// Pin Definition
#define DHTPIN 2
#define DHTTYPE DHT11
#define PIRPIN 3
#define RELAYPIN 4

// Ambang Batas Suhu (Derajat Celsius)
const float AMBANG_SUHU = 28.0;

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  
  pinMode(PIRPIN, INPUT);
  pinMode(RELAYPIN, OUTPUT);
  
  // Standby Relay Off (LOW)
  digitalWrite(RELAYPIN, LOW);
  
  Serial.println("Sistem Kipas Otomatis Siap...");
}

void loop() {
  // Membaca data sensor
  float suhu = dht.readTemperature();
  int adaOrang = digitalRead(PIRPIN);

  if (isnan(suhu)) {
    Serial.println("Gagal membaca dari sensor DHT11!");
    delay(2000);
    return;
  }

  // Log ke Serial Monitor
  Serial.print("Suhu: ");
  Serial.print(suhu);
  Serial.print(" °C | Gerakan: ");
  Serial.println(adaOrang == HIGH ? "TERDETEKSI" : "TIDAK ADA");

  // Logika Kontrol Kipas: (PIR HIGH && SUHU >= AMBANG)
  if (adaOrang == HIGH && suhu >= AMBANG_SUHU) {
    digitalWrite(RELAYPIN, HIGH); // Nyalakan Relay (Kipas ON)
    Serial.println("Status Kipas: ON (Ada orang & Suhu Panas)");
  } else {
    digitalWrite(RELAYPIN, LOW);  // Matikan Relay (Kipas OFF)
    Serial.println("Status Kipas: OFF");
  }

  delay(1000);
}
